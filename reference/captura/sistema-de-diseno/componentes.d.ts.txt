  nombre: string;
  fase?: string | null;

  entregableEstado?: 'desconocido' | 'na';

  unidad?: string;

export interface TarjetaActividadProps extends React.HTMLAttributes<HTMLDivElement> {
  /** Número visible («ACT-04»); null = «—» (sin número). */
  codigo?: string | null;
  /** Obsoleto: se muestra si no hay codigo. */
  id?: string;
  nombre: string; responsable?: string | null; entregable?: string | null;
  estado?: 'confirmado' | 'sugerencia' | 'pregunta';

  arrastrando?: boolean;

  origen?: boolean;
  /** Su número acaba de cambiar: parpadea el anillo. */
  renumerada?: boolean;
  seleccionada?: boolean;
  /** Solo número, nombre y estado (paso «listar y agrupar»). */
  compacto?: boolean;

  /** El nombre se corrige con un clic (lo pone TableroFases). */
  renombrable?: boolean;
  /** Muestra el campo de nombre en su sitio: Enter guarda, Esc cancela. */
  renombrando?: boolean;
  onRenombrar?: (nombre: string) => void;
  onCancelarRenombrar?: () => void;

export declare function TarjetaActividad(props: TarjetaActividadProps): React.ReactElement;
export interface ColumnaFaseProps {
  nombre: string; numero?: number; objetivo?: string;
  entrada?: string | null; entregables?: string[]; salida?: string | null;
  actividades: Array<TarjetaActividadProps & { clave?: string }>;
  /** Columna de actividades aún sin fase: sin número. */
  sinFase?: boolean;
  /** Es la fase donde caerá la tarjeta que se arrastra. */
  destino?: boolean;
  /** Oculta objetivo y criterios: columna del paso «listar y agrupar». */
  compacta?: boolean;

  onEditar?: () => void;
  /** Hace del nombre un botón que pide corregirlo. */
  onPedirRenombrar?: () => void;
  renombrando?: boolean;
  onRenombrar?: (nombre: string) => void;
  onCancelarRenombrar?: () => void;
  /** Debajo de las tarjetas (TableroFases pone «+ Agregar actividad»). */

export interface FaseTablero {
  id: string; nombre: string; objetivo?: string;
  entrada?: string | null; entregables?: string[]; salida?: string | null;
  /** La columna «Sin fase»: sus actividades no se numeran. */
  sinFase?: boolean;
  /** En orden: de arriba abajo es la secuencia. */
  actividades: Array<{ clave: string; nombre: string; responsable?: string | null; entregable?: string | null; estado?: 'confirmado' | 'sugerencia' | 'pregunta' }>;
}
export interface PasoSecuencia { clave: string; nombre: string; numero: number | null; codigo: string | null; fase: string | null; faseNombre: string | null }
export interface TableroFasesProps {
  /** Fases con sus actividades en orden. Si cambia desde fuera, el tablero lo adopta (tras soltar si hay un arrastre). */
  fases: FaseTablero[];
  /** Prefijo del número visible. Por defecto «ACT-». */
  prefijo?: string;
  seleccionada?: string;
  /** Clic fuera del nombre, Enter o «Ficha»: abre el detalle. */
  onSeleccionar?: (clave: string) => void;
  /** Tras cada movimiento: tablero nuevo y secuencia numerada. */
  onCambio?: (fases: FaseTablero[], secuencia: PasoSecuencia[]) => void;
  /** Paso «listar y agrupar»: tarjetas y columnas compactas. */
  compacto?: boolean;
  /** Sin arrastre ni edición (versión en revisión o aprobada). */
  soloLectura?: boolean;
  /** «+ Agregar actividad» al pie de cada columna; varias líneas pegadas = varias actividades. */
  onAgregar?: (faseId: string, nombres: string[]) => void;
  /** Clic en el nombre o F2. */
  onRenombrar?: (clave: string, nombre: string) => void;
  /** Suprimir sobre una tarjeta con foco. */
  onEliminar?: (clave: string) => void;
  /** Columna «+ Nueva fase» antes de «Sin fase». */
  onNuevaFase?: (nombre: string) => void;
  onRenombrarFase?: (id: string, nombre: string) => void;
  /** Enlace «Objetivo y criterios» de cada fase. */
  onEditarFase?: (id: string) => void;
}
export declare function TableroFases(props: TableroFasesProps): React.ReactElement;
/** Fases de izquierda a derecha, actividades de arriba abajo; «Sin fase» no cuenta. */

/** Destino especial de una salida: termina el proceso. */
export declare const DESTINO_FIN: '__fin';
export interface SalidaDecision {
  condicion: string;
  /** Clave de la actividad destino, DESTINO_FIN, o null si aún no se sabe. */
  destino?: string | null;
  /** Obsoleto: texto libre para maquetas. */
  siguiente?: string;
  porDefecto?: boolean;
  clase?: 'retrabajo' | 'rechazo' | 'cancelacion' | 'excepcion';
}
export interface Decision {
  pregunta: string;
  /** Clave de la actividad de la que sale la decisión. */
  origen?: string | null;
  decide?: string;
  tipo?: 'exclusiva' | 'inclusiva' | 'paralela';
  salidas: SalidaDecision[];
  /** Guardado al confirmar: la fase del origen en ese momento. */
  contexto?: { faseOrigen?: string | null };
  /** Ids de motivos que el analista decidió mantener. */
  aceptados?: string[];
}
export interface MotivoRevision {
  id: string;
  tipo: 'origen-eliminado' | 'origen-sin-fase' | 'origen-cambio-fase' | 'destino-eliminado' | 'destino-sin-fase' | 'hacia-atras';
  texto: string;
  /** Índice de la salida afectada, si aplica. */
  salida?: number;
  /** Se puede silenciar con «Mantener así». */
  mantenible: boolean;
}
/** Nunca borra nada: devuelve por qué la decisión necesita revisión. */
export declare function revisarDecision(decision: Decision, secuencia: PasoSecuencia[]): MotivoRevision[];
export interface TarjetaDecisionProps extends Decision {

  secuencia?: PasoSecuencia[];
  /** Motivos ya calculados; si falta, se calculan con revisarDecision. */
  motivos?: MotivoRevision[];
  onMantener?: (ids: string[]) => void;
  onReasignarOrigen?: () => void;
  onEliminar?: () => void;
  onMarcarRetrabajo?: (salida: number) => void;
  onCambiarDestino?: (salida: number) => void;
  onEditar?: () => void;
}
export declare function TarjetaDecision(props: TarjetaDecisionProps): React.ReactElement;

/** Primitiva SVG de un ícono en un cuadro de 16×16: etiqueta, atributos, si va rellena y si su trazo va en claro (sobre relleno). */

/** La geometría de los íconos, compartida por la interfaz, el lienzo BPMN y el documento Word. Trazo 1,4, puntas redondas. */
export declare const ICONOS_BPMN: Record<NombreIconoBPMN, PrimitivaIcono[]>;
export interface IconoBPMNProps {
  nombre: NombreIconoBPMN;
  /** Lado en px. Por defecto 16. */
  tamano?: number;
  /** Nombre accesible. true usa el nombre en español del ícono; sin título el ícono es decorativo (aria-hidden) y debe ir con texto. */
  titulo?: string | true;
  className?: string;
}
/** Un ícono suelto: hereda el color del texto (currentColor). */
export declare function IconoBPMN(props: IconoBPMNProps): React.ReactElement | null;
